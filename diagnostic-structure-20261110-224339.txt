    )

            Count = 0

    }

        $firstFields = ''

            $Value[0] -is [System.Management.Automation.PSCustomObject]) {
            $firstFields = (
                @($Value[0].PSObject.Properties | ForEach-Object {
                    $_.Name
                }) -join ', '
            )

            Path = $Path
            Kind = 'Array'
            Count = $Value.Count
            Members = $firstFields
        }

            if ($Value[0] -is [System.Management.Automation.PSCustomObject] -or
                $Value[0] -is [System.Array]) {

                Get-JsonShape -Value $Value[0] `
                    -Path "$Path[0]" `
                    -Depth ($Depth + 1)
            }
        }

        return
    }

    if ($Value -is [System.Management.Automation.PSCustomObject]) {
        $properties = @($Value.PSObject.Properties)

        [PSCustomObject]@{
            Path = $Path
            Kind = 'Object'
            Count = $properties.Count
            Members = ($properties.Name -join ', ')
        }

        if ($Depth -lt 3) {
            foreach ($property in $properties) {
                if ($property.Value -is [System.Array] -or
                    $property.Value -is [System.Management.Automation.PSCustomObject]) {

                    Get-JsonShape -Value $property.Value `
                        -Path "$Path.$($property.Name)" `
                        -Depth ($Depth + 1)
                }
            }
        }

        return
    }

    [PSCustomObject]@{
        Path = $Path
        Kind = $Value.GetType().Name
        Count = 1
        Members = ''
    }
}
